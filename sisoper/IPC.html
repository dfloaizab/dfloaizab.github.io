1. ¿Por qué IPC? Mapa de mecanismos

Cada proceso tiene su propio espacio de direcciones: no puede leer ni escribir la memoria de otro. Cuando los procesos deben cooperar (modularidad, paralelismo, compartir información) el SO ofrece mecanismos de comunicación entre procesos. Hay dos modelos conceptuales: paso de mensajes (el kernel copia los datos: pipes, FIFOs, colas, sockets) y memoria compartida (rápida, pero exige sincronización explícita).

2. Pipes anónimos

Concepto

pipe(int fd[2]) crea un buffer en el kernel (64 KiB por defecto en Linux) con dos descriptores: fd[0] lectura y fd[1] escritura. Tras fork() el hijo hereda ambos; cada proceso debe cerrar el extremo que no usa.

  • read bloquea si el pipe está vacío; devuelve 0 (EOF) cuando todos los extremos de escritura están cerrados.
  • write bloquea si está lleno; si no hay lectores recibe SIGPIPE.
  • Escrituras ≤ PIPE_BUF (4096 B) son atómicas.
  • Con dup2 se redirige stdin/stdout: así funciona ls | wc -l en el shell.
Error típico: olvidar cerrar fd[1] en el lector: nunca llega el EOF y el programa se cuelga.
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>

int main(void) {
    int fd[2];
    if (pipe(fd) == -1) { perror("pipe"); return 1; }
    pid_t pid = fork();
    if (pid == 0) {                 /* HIJO: lector */
        close(fd[1]);
        char buf[64]; ssize_t n;
        while ((n = read(fd[0], buf, sizeof buf - 1)) > 0) {
            buf[n] = '\0';
            printf("Hijo recibio: %s\n", buf);
        }
        close(fd[0]); _exit(0);
    }
    close(fd[0]);                   /* PADRE: escritor */
    const char *m = "Hola desde el padre";
    write(fd[1], m, strlen(m));
    close(fd[1]);                   /* provoca EOF en el hijo */
    wait(NULL);
    return 0;
}

3. FIFOs (pipes con nombre)

Un FIFO es un pipe con entrada en el sistema de archivos; permite comunicar procesos sin parentesco. open() bloquea hasta que otro proceso abre el extremo opuesto (salvo O_NONBLOCK). Se crea con mkfifo("/tmp/lab_fifo", 0600) o desde la terminal con mkfifo, y se elimina con unlink.

/* escritor.c */                          /* lector.c */
mkfifo("/tmp/lab_fifo", 0600);            int fd = open("/tmp/lab_fifo", O_RDONLY);
int fd = open("/tmp/lab_fifo", O_WRONLY); char buf[128]; ssize_t n;
write(fd, "ping", 4);                     while ((n = read(fd, buf, sizeof buf)) > 0)
close(fd);                                    write(STDOUT_FILENO, buf, n);
                                          close(fd); unlink("/tmp/lab_fifo");

4. Señales

Concepto

Una señal es una interrupción por software asíncrona. Útiles: SIGINT (Ctrl+C), SIGTERM (terminación cortés), SIGKILL/SIGSTOP (no capturables), SIGCHLD (un hijo terminó), SIGUSR1/2 (definidas por el usuario), SIGPIPE, SIGALRM.

  • Use sigaction y no signal (semántica portable y definida).
  • En el manejador solo funciones async-signal-safe (write, _exit); no printf ni malloc.
  • Comunique con el programa principal mediante volatile sig_atomic_t.
  • Las señales estándar no se encolan: varias iguales pendientes se fusionan en una.
#include <signal.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdio.h>

static volatile sig_atomic_t recibida = 0;
static void manejador(int s) { (void)s; recibida = 1; }

int main(void) {
    struct sigaction sa;
    sa.sa_handler = manejador;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    sigaction(SIGUSR1, &sa, NULL);   /* antes del fork */

    pid_t pid = fork();
    if (pid == 0) {
        while (!recibida) pause();
        write(STDOUT_FILENO, "Hijo: SIGUSR1 ok\n", 17);
        _exit(0);
    }
    sleep(1);
    kill(pid, SIGUSR1);
    wait(NULL);
    return 0;
}

Hay una carrera sutil entre while(!recibida) y pause(). Actividad avanzada: corríjala con sigprocmask + sigsuspend.

5. Colas de mensajes POSIX

Mensajes delimitados (no flujo de bytes) con prioridad: mq_receive entrega primero el de mayor prioridad. Los nombres empiezan con /. El buffer de recepción debe ser ≥ mq_msgsize. Se listan en /dev/mqueue (si está montado). Enlace con -lrt.

#include <mqueue.h>
#include <fcntl.h>
#include <stdio.h>

int main(void) {
    struct mq_attr at = {0};
    at.mq_maxmsg = 8; at.mq_msgsize = 64;
    mqd_t q = mq_open("/cola_lab", O_CREAT | O_RDWR, 0600, &at);
    if (q == (mqd_t)-1) { perror("mq_open"); return 1; }

    mq_send(q, "baja", 5, 1);
    mq_send(q, "URGENTE", 8, 9);

    char buf[64]; unsigned prio;
    for (int i = 0; i < 2; i++) {
        mq_receive(q, buf, sizeof buf, &prio);
        printf("prio=%u msg=%s\n", prio, buf);   /* URGENTE primero */
    }
    mq_close(q); mq_unlink("/cola_lab");
    return 0;
}

6. Memoria compartida + semáforos

El mecanismo más rápido: tras mmap con MAP_SHARED, ambos procesos ven la misma memoria física sin copias del kernel. No hay sincronización automática: se necesitan semáforos (con pshared=1 si viven dentro de la región). Patrón clásico: productor-consumidor con buffer acotado.

#define N 5
typedef struct { sem_t vacio, lleno, mutex; int buf[N]; int in, out; } Compartido;

/* Creación (proceso padre) */
int fd = shm_open("/shm_lab", O_CREAT | O_RDWR, 0600);
ftruncate(fd, sizeof(Compartido));
Compartido *c = mmap(NULL, sizeof *c, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
sem_init(&c->vacio, 1, N);  sem_init(&c->lleno, 1, 0);  sem_init(&c->mutex, 1, 1);

/* Productor */                          /* Consumidor */
sem_wait(&c->vacio);                      sem_wait(&c->lleno);
sem_wait(&c->mutex);                      sem_wait(&c->mutex);
c->buf[c->in] = dato;                      dato = c->buf[c->out];
c->in = (c->in + 1) % N;                   c->out = (c->out + 1) % N;
sem_post(&c->mutex);                      sem_post(&c->mutex);
sem_post(&c->lleno);                      sem_post(&c->vacio);

/* Limpieza */
munmap(c, sizeof *c); close(fd); shm_unlink("/shm_lab");
Orden de los semáforos: si el productor toma mutex antes que vacio y el buffer está lleno, se produce un interbloqueo (conecta con la unidad de deadlocks).

7. Sockets UNIX (dominio local)

socketpair(AF_UNIX, SOCK_STREAM, 0, sv) crea dos extremos bidireccionales conectados (ideal padre-hijo). Para procesos independientes: socket + bind a una ruta (struct sockaddr_un) + listen/accept y connect. Además permiten pasar descriptores de archivo entre procesos (SCM_RIGHTS).

int sv[2];
socketpair(AF_UNIX, SOCK_STREAM, 0, sv);
if (fork() == 0) {                       /* hijo */
    close(sv[0]);
    char b[32]; ssize_t n = read(sv[1], b, sizeof b);
    write(sv[1], "eco:", 4); write(sv[1], b, n);
    _exit(0);
}
close(sv[1]);                            /* padre */
write(sv[0], "hola", 4);
char r[64]; ssize_t n = read(sv[0], r, sizeof r);
write(STDOUT_FILENO, r, n); wait(NULL);

8. Criterios de selección y errores frecuentes

¿Cuál uso?

  • Padre-hijo, flujo simple → pipe.
  • Procesos independientes, unidireccional → FIFO.
  • Mensajes discretos con prioridad → cola POSIX.
  • Grandes volúmenes, baja latencia → memoria compartida + semáforos.
  • Diálogo bidireccional / cliente-servidor local → socket UNIX.
  • Solo avisar un evento → señal.

Errores y diagnóstico

  • No cerrar extremos sin uso → bloqueos y EOF que no llega.
  • No verificar retornos (-1 + perror/errno); ignorar lecturas parciales.
  • Olvidar unlink/mq_unlink/shm_unlink → recursos huérfanos.
  • Procesos zombi por no hacer wait.
  • Condiciones de carrera por falta de exclusión mutua.
